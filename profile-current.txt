import Link from "next/link";
import { notFound } from "next/navigation";

import FollowButton from "@/components/FollowButton";
import Header from "@/components/Header";

import PostCard, {
  type SupabasePostCard,
} from "@/components/PostCard";

import Sidebar from "@/components/Sidebar";

import { createClient } from "@/lib/supabase/server";

type Props = {
  params: Promise<{
    username: string;
  }>;

  searchParams: Promise<{
    tab?: string;
  }>;
};

type Profile = {
  id: string;
  username: string;
  display_name: string | null;
  avatar_url: string | null;
  bio: string | null;
  created_at: string;
};

type FollowState = {
  follower_count:
    | number
    | string;

  following_count:
    | number
    | string;

  is_following:
    boolean;
};

type RepostRow = {
  created_at: string;

  post:
    | SupabasePostCard
    | SupabasePostCard[]
    | null;
};

function getSinglePost(
  value:
    | SupabasePostCard
    | SupabasePostCard[]
    | null,
) {
  if (
    Array.isArray(value)
  ) {
    return (
      value[0] ??
      null
    );
  }

  return value;
}

function formatCount(
  value:
    | number
    | string
    | null
    | undefined,
) {
  const number =
    Number(value ?? 0);

  if (
    !Number.isFinite(
      number,
    )
  ) {
    return "0";
  }

  return new Intl
    .NumberFormat(
      "en-US",
    )
    .format(number);
}

export default async function ProfilePage({
  params,
  searchParams,
}: Props) {
  const {
    username,
  } = await params;

  const {
    tab,
  } = await searchParams;

  const activeTab =
    tab === "reposts"
      ? "reposts"
      : "posts";

  const decodedUsername =
    decodeURIComponent(
      username,
    ).trim();

  const supabase =
    await createClient();

  /*
   * -----------------------------------------
   * Current viewer
   * -----------------------------------------
   */

  const {
    data: {
      user:
        currentUser,
    },
  } =
    await supabase.auth
      .getUser();

  /*
   * -----------------------------------------
   * Profile
   * -----------------------------------------
   */

  const {
    data:
      profileData,
    error:
      profileError,
  } =
    await supabase
      .from("profiles")
      .select(`
        id,
        username,
        display_name,
        avatar_url,
        bio,
        created_at
      `)
      .ilike(
        "username",
        decodedUsername,
      )
      .maybeSingle();

  if (
    profileError ||
    !profileData
  ) {
    if (profileError) {
      console.error(
        "Profile lookup failed:",
        profileError,
      );
    }

    notFound();
  }

  const profile =
    profileData as Profile;

  const isOwnProfile =
    currentUser?.id ===
    profile.id;

  /*
   * -----------------------------------------
   * Followers / Following
   * -----------------------------------------
   */

  const {
    data:
      followStateData,
    error:
      followStateError,
  } =
    await supabase.rpc(
      "get_user_follow_state",
      {
        p_user_id:
          profile.id,
      },
    );

  if (
    followStateError
  ) {
    console.error(
      "Follow state failed:",
      followStateError,
    );
  }

  const followState =
    (
      followStateData?.[0] as
        | FollowState
        | undefined
    ) ?? {
      follower_count: 0,
      following_count: 0,
      is_following: false,
    };

  /*
   * -----------------------------------------
   * Original posts
   * -----------------------------------------
   */

  const {
    data:
      postsData,
    error:
      postsError,
  } =
    await supabase
      .from("posts")
      .select(`
        id,
        author_id,
        title,
        description,
        media_url,
        media_type,
        verification_status,
        verification_number,
        real_vote_count,
        ai_vote_count,
        comment_count,
        repost_count,
        created_at,

        author:profiles!posts_author_id_fkey (
          username
        ),

        board:boards!posts_board_id_fkey (
          name,
          slug
        )
      `)
      .eq(
        "author_id",
        profile.id,
      )
      .eq(
        "status",
       "published",
      )
      .order(
        "created_at",
        {
          ascending:
            false,
        },
      )
      .limit(100);

  if (postsError) {
    console.error(
      "Profile posts failed:",
      postsError,
    );
  }

  const posts =
    (postsData ??
      []) as unknown as
      SupabasePostCard[];

  /*
   * -----------------------------------------
   * Reposts
   * -----------------------------------------
   */

  const {
    data:
      repostData,
    error:
      repostError,
  } =
    await supabase
      .from(
        "post_reposts",
      )
      .select(`
        created_at,

        post:posts!post_reposts_post_id_fkey (
          id,
          author_id,
          title,
          description,
          media_url,
          media_type,
          verification_status,
          verification_number,
          real_vote_count,
          ai_vote_count,
          comment_count,
          repost_count,
          created_at,

          author:profiles!posts_author_id_fkey (
            username
          ),

          board:boards!posts_board_id_fkey (
            name,
            slug
          )
        )
      `)
      .eq(
        "user_id",
        profile.id,
      )
      .order(
        "created_at",
        {
          ascending:
            false,
        },
      )
      .limit(100);

  if (repostError) {
    console.error(
      "Profile reposts failed:",
      repostError,
    );
  }

  const reposts =
    (repostData ??
      []) as unknown as
      RepostRow[];

  const profileName =
    profile.display_name?.trim() ||
    profile.username;

  const profileInitial =
    profileName
      .slice(0, 1)
      .toUpperCase();

  return (
    <main className="min-h-screen bg-[#0b0d10] text-[#e7e9ea]">
      <Header />

      <div className="mx-auto flex w-full max-w-[1400px] gap-6 px-4 py-6 lg:px-6">
        <aside className="hidden w-[250px] shrink-0 lg:block">
          <Sidebar />
        </aside>

        <section className="min-w-0 flex-1">
          <div className="overflow-hidden rounded-2xl border border-white/10 bg-[#12151a]">
            <div className="h-28 bg-gradient-to-r from-[#142b40] via-[#15212e] to-[#171b22]" />

            <div className="px-5 pb-6 sm:px-7">
              <div className="-mt-12 flex flex-col gap-5 sm:flex-row sm:items-end">
                {profile.avatar_url ? (
                  // eslint-disable-next-line @next/next/no-img-element
                  <img
                    src={
                      profile.avatar_url
                    }
                    alt={`${profileName} profile`}
                    className="h-24 w-24 shrink-0 rounded-full border-4 border-[#12151a] bg-[#12151a] object-cover"
                  />
                ) : (
                  <div className="flex h-24 w-24 shrink-0 items-center justify-center rounded-full border-4 border-[#12151a] bg-[#17304a] text-4xl font-black text-[#69b7ff]">
                    {
                      profileInitial
                    }
                  </div>
                )}

                <div className="flex min-w-0 flex-1 flex-wrap items-end justify-between gap-4 pb-1">
                  <div className="min-w-0">
                    <h1 className="break-words text-3xl font-black text-white">
                      {
                        profileName
                      }
                    </h1>

                    <p className="mt-1 text-sm text-white/40">
                      @
                      {
                        profile.username
                      }
                    </p>
                  </div>

                  {!isOwnProfile && (
                    <FollowButton
                      userId={
                        profile.id
                      }
                      initialFollowing={
                        Boolean(
                          followState.is_following,
                        )
                      }
                    />
                  )}
                </div>
              </div>

              {profile.bio && (
                <p className="mt-5 max-w-2xl whitespace-pre-wrap text-sm leading-6 text-white/65">
                  {
                    profile.bio
                  }
                </p>
              )}

              <div className="mt-5 flex flex-wrap items-center gap-5 text-sm">
                <Link
  href={`/profile/${encodeURIComponent(
    profile.username,
  )}/followers`}
  className="text-white/45 transition hover:text-white"
>
  <strong className="mr-1 font-black text-white">
    {formatCount(
      followState.follower_count,
    )}
  </strong>
  Followers
</Link>

<Link
  href={`/profile/${encodeURIComponent(
    profile.username,
  )}/following`}
  className="text-white/45 transition hover:text-white"
>
  <strong className="mr-1 font-black text-white">
    {formatCount(
      followState.following_count,
    )}
  </strong>
  Following
</Link>
              </div>
            </div>
          </div>

          <div className="mt-5 flex border-b border-white/10">
            <Link
              href={`/profile/${encodeURIComponent(
                profile.username,
              )}`}
              className={`relative px-5 py-4 text-sm font-black transition ${
                activeTab ===
                "posts"
                  ? "text-white"
                  : "text-white/35 hover:text-white/70"
              }`}
            >
              Posts

              {activeTab ===
                "posts" && (
                <span className="absolute inset-x-4 bottom-0 h-0.5 rounded-full bg-[#48a7ff]" />
              )}
            </Link>

            <Link
              href={`/profile/${encodeURIComponent(
                profile.username,
              )}?tab=reposts`}
              className={`relative px-5 py-4 text-sm font-black transition ${
                activeTab ===
                "reposts"
                  ? "text-white"
                  : "text-white/35 hover:text-white/70"
              }`}
            >
              Reposts

              {activeTab ===
                "reposts" && (
                <span className="absolute inset-x-4 bottom-0 h-0.5 rounded-full bg-[#48a7ff]" />
              )}
            </Link>
          </div>

          {activeTab ===
          "posts" ? (
            <div className="mt-5 space-y-4">
              {posts.length ===
              0 ? (
                <div className="rounded-xl border border-white/10 bg-[#12151a] px-6 py-16 text-center">
                  <p className="text-sm font-semibold text-white/35">
                    No posts yet.
                  </p>
                </div>
              ) : (
                posts.map(
                  (
                    post,
                  ) => (
                    <PostCard
                      key={
                        post.id
                      }
                      post={
                        post
                      }
                    />
                  ),
                )
              )}
            </div>
          ) : (
            <div className="mt-5 space-y-4">
              {reposts.length ===
              0 ? (
                <div className="rounded-xl border border-white/10 bg-[#12151a] px-6 py-16 text-center">
                  <p className="text-sm font-semibold text-white/35">
                    No reposts yet.
                  </p>
                </div>
              ) : (
                reposts.map(
                  (
                    row,
                    index,
                  ) => {
                    const post =
                      getSinglePost(
                        row.post,
                      );

                    if (
                      !post
                    ) {
                      return null;
                    }

                    return (
                      <div
                        key={`${post.id}-${row.created_at}-${index}`}
                      >
                        <div className="mb-2 flex items-center gap-2 px-2 text-xs font-bold text-white/35">
                          <svg
                            viewBox="0 0 24 24"
                            aria-hidden="true"
                            className="h-4 w-4"
                            fill="none"
                            stroke="currentColor"
                            strokeWidth="1.8"
                            strokeLinecap="round"
                            strokeLinejoin="round"
                          >
                            <path d="m17 2 4 4-4 4" />
                            <path d="M3 11V9a3 3 0 0 1 3-3h15" />
                            <path d="m7 22-4-4 4-4" />
                            <path d="M21 13v2a3 3 0 0 1-3 3H3" />
                          </svg>

                          <span>
                            @
                            {
                              profile.username
                            }{" "}
                            reposted
                          </span>
                        </div>

                        <PostCard
                          post={
                            post
                          }
                        />
                      </div>
                    );
                  },
                )
              )}
            </div>
          )}
        </section>
      </div>
    </main>
  );
}
